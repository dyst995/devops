# Commands to memorize

```bash
# sha-bang is line 1 of the file (examples from the notes)
#!/bin/sh
#!/bin/bash
#!/usr/bin/perl
#!/usr/bin/env python
#!/bin/sed -f
#!/bin/awk -f

sh scriptname                # run the file with sh; execute bit not required
bash scriptname              # run the file with bash; execute bit not required

chmod +x scriptname          # make the script directly executable
./scriptname                 # run it; kernel uses the sha-bang interpreter
```
